# -- Background jobs (Celery) --------------------------------------------------
# celery: runs the periodic jobs in backend/celery.py (scheduled admin
#   notification dispatch, auto-archiving expired exams). Needs a separate
#   `celery -A backend worker` process plus `celery -A backend beat` in
#   production (see Procfile) - it does nothing extra by just being installed.
# redis: the message broker celery connects to, via CELERY_BROKER_URL /
#   REDIS_URL. django-celery-beat was deliberately NOT added - the current
#   release forces a Django downgrade incompatible with the Django==6.1 pin
#   above, so the beat schedule lives as a plain CELERY_BEAT_SCHEDULE dict in
#   settings.py instead of an admin-editable DB table.
celery==5.6.3
redis==8.1.0

# import (administration/import_views.py) and generates the downloadable
# .xlsx template/error report.
openpyxl==3.1.5

# -- PDF generation ------------------------------------------------------------
# Creates printable subjective examination question papers from the approved
# Question Bank and stores them through the Examination FileField storage.
reportlab==5.0.1
